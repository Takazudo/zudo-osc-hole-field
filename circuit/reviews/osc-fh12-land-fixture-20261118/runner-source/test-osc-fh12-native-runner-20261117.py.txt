    def setUp(self):
        self.tmp=tempfile.TemporaryDirectory();self.addCleanup(self.tmp.cleanup)

    def commands(self,failure=None,remaining=False,worker_live=False):
        calls=[]
        def run(args,timeout=8):
            calls.append(args)
            if failure and failure in args:raise RuntimeError('injected '+failure)
            if args[0]=='systemctl':
                if 'show' in args:
                    return 'ActiveState=active\nMainPID=42\nControlGroup=/live' if worker_live else 'ActiveState=inactive\nMainPID=0\nControlGroup='
                return ''
            if 'ps' in args:return 'ownedid' if remaining or len([c for c in calls if 'ps' in c])==1 else ''
            if 'inspect' in args:return json.dumps([{'Name':'/'+self.name,'Config':{'Labels':{'osc.fixture.owner':self.name}}}])
            return ''
        return run,calls
    def test_success_cleanup_proves_container_absence(self):
        run,calls=self.commands();self.assertEqual(r.cleanup(self.run_dir,run),[])
        self.assertTrue(any('rm' in c for c in calls));self.assertFalse((self.run_dir/'ADMISSION_BLOCKED.json').exists())
    def test_start_failure_still_cleans_container(self):
        # Failed worker startup/stop must not suppress independent Docker cleanup.
        run,calls=self.commands(failure='stop');self.assertTrue(r.cleanup(self.run_dir,run))
        self.assertTrue(any('rm' in c for c in calls));self.assertTrue((self.run_dir/'ADMISSION_BLOCKED.json').exists())
    def test_container_removal_failure_never_passes(self):
        run,calls=self.commands(failure='rm');self.assertTrue(r.cleanup(self.run_dir,run))
    def test_container_absence_unverified_never_passes(self):
        run,calls=self.commands(remaining=True);self.assertTrue(r.cleanup(self.run_dir,run))
    def test_host_worker_live_never_passes(self):
        run,calls=self.commands(worker_live=True);self.assertTrue(r.cleanup(self.run_dir,run))
        self.assertTrue(any('rm' in c for c in calls))
    def test_receipt_write_failure_still_attempts_daemon_cleanup(self):

    def test_abrupt_controller_failure_has_independent_cleanup_under_lock(self):
        lock=self.cache/'synthetic-shared.lock';holder=lock.open('w');fcntl.flock(holder,fcntl.LOCK_EX)
        cleanup=r.cleanup;finalize=r.finalize_controller;removed=[]
        def assert_locked():
            with lock.open('a') as contender:
                with self.assertRaises(BlockingIOError):fcntl.flock(contender,fcntl.LOCK_EX|fcntl.LOCK_NB)
        def fake(args,timeout=8):
            assert_locked()
            if args[0]=='systemd-run':return ''
            if args[0]=='systemctl':
                if 'show' not in args:return ''
                if 'SubState' in args:return 'ActiveState=failed\nSubState=failed\nExecMainStatus=9\nExecMainCode=2'
                if '--value' in args:return 'inactive'
                return 'ActiveState=inactive\nMainPID=0\nControlGroup=\nResult=oom-kill\nExecMainStatus=9'
            if 'ps' in args:return '' if removed else 'ownedid'
            if 'inspect' in args:return json.dumps([{'Name':'/'+self.name,'Config':{'Labels':{'osc.fixture.owner':self.name}}}])
            if 'rm' in args:removed.append(True)
            return ''
        with mock.patch.object(r,'verify_runner_sources'),mock.patch.object(r,'guard_owner',return_value=os.getpid()),mock.patch.object(r,'effective_caps',return_value={}),mock.patch.object(r.resource,'setrlimit'),mock.patch.object(r.signal,'signal'),mock.patch.object(r,'command',side_effect=fake),mock.patch.object(r,'finalize_controller',side_effect=lambda path,deadline=None:finalize(path,run=fake,deadline=deadline)),mock.patch.object(r,'cleanup',side_effect=lambda path,deadline=None:cleanup(path,run=fake,deadline=deadline)):
            self.assertEqual(r.guardian(self.run_dir,'identity'),1)
        self.assertTrue(removed);assert_locked()
        receipt=json.loads((self.run_dir/'supervisor.json').read_text())
        self.assertEqual(receipt['cleanup'],'VERIFIED');self.assertEqual(receipt['status'],'FAILED')
        fcntl.flock(holder,fcntl.LOCK_UN);holder.close()

    def export_frame(self,entries,body=b'',total=None):
        header={'protocol':'OSC_FIXTURE_EXPORT_1','entries':entries,'total_size':sum(e['size'] for e in entries) if total is None else total}
        raw=json.dumps(header,separators=(',',':')).encode()
        return b'OSC_FIXTURE_EXPORT_1\n'+struct.pack('!I',len(raw))+raw+body+b'OSC_FIXTURE_EXPORT_END\n'
    def file_entry(self,name,data=b'data'):
        import hashlib
        return {'path':name,'type':'file','size':len(data),'sha256':hashlib.sha256(data).hexdigest()}
    def test_binary_export_preserves_failure_payload(self):
        body=b'{"status":"FAILED","traceback":"real detail"}'
        entries=[self.file_entry('transaction-failure.json',body)]
        output=self.run_dir/'decoded';output.mkdir()
        r.decode_export(io.BytesIO(self.export_frame(entries,body)),output)
        self.assertEqual((output/'transaction-failure.json').read_bytes(),body)
    def test_export_truncation_and_bad_digest_are_refused(self):
        entry=self.file_entry('payload');framed=self.export_frame([entry],b'data')
        for i,blob in enumerate((framed[:-1],framed[:-25],framed.replace(b'data',b'evil'))):
            output=self.run_dir/('bad'+str(i));output.mkdir()
            with self.assertRaises(ValueError):r.decode_export(io.BytesIO(blob),output)
    def test_export_rejects_unsafe_paths_and_types(self):
        cases=[self.file_entry(n) for n in ('../escape','/absolute','a/../escape','a//b','./b','a\\b')]
        for kind in ('symlink','hardlink','device','fifo'):
            entry=self.file_entry('bad');entry['type']=kind;cases.append(entry)
        for i,entry in enumerate(cases):
            output=self.run_dir/('unsafe'+str(i));output.mkdir()
            with self.assertRaises(ValueError):r.decode_export(io.BytesIO(self.export_frame([entry],b'data')),output)
    def test_export_size_count_and_header_bounds(self):

    def test_export_timeout_is_bounded_and_recorded(self):
        read,write=os.pipe()
        try:
            with os.fdopen(read,'rb',buffering=0) as stream:
                timed=r.DeadlinePipe(stream,time.monotonic()-1)
                try:
                    with self.assertRaises(TimeoutError):timed.read(1)
                finally:timed.close()
        finally:os.close(write)
        def fail(name,output):raise TimeoutError('fixture timeout')
        with self.assertRaises(TimeoutError):r.capture_output(self.run_dir,self.name,'timeout',fail)
        self.assertEqual(json.loads((self.run_dir/'timeout-export-failure.json').read_text())['status'],'EXPORT_FAILED')
