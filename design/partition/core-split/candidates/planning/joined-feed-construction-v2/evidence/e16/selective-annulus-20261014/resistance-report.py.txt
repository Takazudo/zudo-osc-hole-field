Run from the repository root: python3 PATH/TO/resistance-report.py.txt
Writes only resistance-*.json/csv under this helper's directory.
"""
import collections, csv, gzip, hashlib, json, math
from pathlib import Path
OUT = Path(__file__).resolve().parent
E = OUT.parent
ROOT = next(p for p in OUT.parents if (p / 'circuit/WORKFLOW.md').exists())
CACHE = Path('/home/takazudo/.cache/zudo-e16-runs/coupled-approved-852ffac-8262c033')
inputs = []
def read(path, expected=None):
    raw = path.read_bytes()
    digest = hashlib.sha256(raw).hexdigest()
    if expected is not None:
        assert digest == expected, (str(path), digest, expected)
    inputs.append({'path': str(path.relative_to(ROOT)) if path.is_relative_to(ROOT) else str(path), 'size_bytes': len(raw), 'sha256': digest, 'matches_retained_binding': expected is not None})
    return json.loads(gzip.decompress(raw) if path.suffix == '.gz' else raw)
s = read(E/'campaign-wsl-20261014/result-summary/attempt5-final/summary.json', '8df092b5f3e6a9a1e28f3f09ebfc17e965b7379ec4861796221eb5c7849a03cd')
assert s['source_sha'] == '852ffac7f85aff4bb436eb25f7ce993cee48146b'
assert not s['integrity_errors'] and not s['failed_numerical_checks']
service = read(CACHE/'device-service-normal-typ-v1.json', s['service']['output_sha256'])
ledger = read(E/'continuation-20261013b/load-ledger.json.gz', service['bindings']['ledger'])
assert ledger['ledger_sha256'] == s['scenario']['ledger_sha256']
rows_by_pad = {r['pad_id']: r for r in ledger['records'] if r['scenario_id']=='normal-typ-v1' and r['pad_id'] is not None}
devices = {(d['board'],d['ref']): d for d in service['device_service']['devices']}
loaded = []; maxima = []; energy = []; return_sensitivity = []
for c in s['coupled']:
    raw = read(Path(c['path']), c['output_sha256'])
    assert raw['bindings']['service'] == s['service']['output_sha256']
    assert (raw['net'],raw['case'],raw['depth']) == (c['net'],c['case'],c['depth'])
    for state, st in raw['states'].items():
        for board, dp in st['driving_point'].items():
            budget = s['budgets']['agnd_copper_ohm' if c['net']=='AGND' else 'rail_copper_ohm']
            assert len(dp['pads']) == len(dp['R_copper_ohm']) == len(dp['R_driving_point_ohm'])
            assert len(dp['pads']) == len(set(dp['pads']))
            for pad, copper, driving in zip(dp['pads'],dp['R_copper_ohm'],dp['R_driving_point_ohm']):
                if pad not in raw['loaded_pads']: continue
                r = rows_by_pad[pad]; d = devices[board,r['ref']]
                assert d['eligible_for_solve'] and d['service_status']=='supported'
                assert any(p['pad_id']==pad and p['served'] for p in d['required_pins'])
                assert copper > budget
                loaded.append(dict(board=board, net=c['net'], ref=r['ref'], pin=r['pad'], pad_id=pad, package=d['symbol'], service=d['service_status'], mapping=r['mapping_status'], current_A=r['current_A'], case=c['case'], state=state, allowance_mohm=1000*budget, R_copper_mohm=1000*copper, exceedance_ratio=copper/budget, R_driving_point_mohm=1000*driving, source_relative_potential_uV=1e6*st['boards'][board]['pad_potential_V'][pad]))
            v=c['states'][state]['boards'][board]
            maxima.append({'board':board,'net':c['net'],'case':c['case'],'state':state,'budget_mohm':1000*budget,'all_evaluated':v['all_evaluated_pads'],'loaded':v['energized_pads']})
            en=st['boards'][board]['energy_W']; copper_energy=sum(en[k] for k in ['barrels','face_copper','inner_copper'])
            energy.append({'board':board,'net':c['net'],'case':c['case'],'state':state,'energy_W':en,'copper_energy_W':copper_energy,'fraction_of_board_copper':{k:en[k]/copper_energy for k in ['barrels','face_copper','inner_copper']}})
        if c['net']=='AGND' and state!='all-present':
            for board in ['KA','KB']:
                base=raw['states']['all-present']['driving_point'][board]
                altered=st['driving_point'][board]
                assert base['pads']==altered['pads']
                changes=[(new-old,pad) for pad,old,new in zip(base['pads'],base['R_copper_ohm'],altered['R_copper_ohm']) if pad in raw['loaded_pads']]
                delta,pad=max(changes)
                return_sensitivity.append({'board':board,'case':c['case'],'state':state,'maximum_loaded_same_pad_copper_delta_mohm':1000*delta,'pad_id':pad})
loaded.sort(key=lambda r:(-r['exceedance_ratio'],r['board'],r['pad_id'],r['case'],r['state']))
assert len(loaded)==1536
