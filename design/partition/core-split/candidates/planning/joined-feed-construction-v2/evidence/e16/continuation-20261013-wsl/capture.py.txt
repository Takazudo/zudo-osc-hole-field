import json,subprocess,time,sys,resource
from pathlib import Path
root=Path('/tmp/osc-wsl-20261013');name=sys.argv[1];cmd=sys.argv[2:]

 p=subprocess.run(cmd,stdout=out,stderr=subprocess.STDOUT)

(root/(name+'.json')).write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r));print((root/(name+'.txt')).read_text()[-5000:]);sys.exit(p.returncode)
