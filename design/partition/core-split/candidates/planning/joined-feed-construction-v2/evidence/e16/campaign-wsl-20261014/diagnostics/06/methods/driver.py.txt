"""One profiled production-input unit; uses the unchanged Ledger and UnitGuard."""
from pathlib import Path
import argparse
import json
import os
import sys

ROOT = Path(__file__).resolve().parents[2]
SOURCE = ROOT / 'design/partition/core-split/candidates/planning/joined-feed-construction-v2'
sys.path.insert(0, str(SOURCE))
import replay_steps as rs
from numerical_unit_guard import UnitGuard

p = argparse.ArgumentParser()
p.add_argument('--spec', type=Path, required=True)
p.add_argument('--output', type=Path, required=True)
p.add_argument('--unit-heavy-guard-sha256', required=True)
a = p.parse_args()
s = json.loads(a.spec.read_text())
assert s['maximum_diagnostic_units'] == 1
assert (s['phase'], s['board'], s['net'], s['case'], s['depth'], s['start'], s['stop']) == (
    'dpoint', 'KB', 'AGND', 'nominal-average', 'uniform-dielectric', 0, 16)
assert 'E16_PROFILE_SCOPE' not in os.environ and 'E16_SOLVE_SCOPE' not in os.environ
assert a.output.is_absolute() and not a.output.is_relative_to(ROOT) and not a.output.exists()
for path, digest in s['input_sha256'].items():
    assert rs.sha(path) == digest, path
guard = UnitGuard(a.unit_heavy_guard_sha256)
guard.check()
a.output.mkdir()
ledger = rs.Ledger(a.output, False, sys.executable, unit_guard=guard)
probe = Path(__file__).with_name('profile_unit.py')
result = a.output / 'result.pkl.gz'
ledger.run('profile-production-dpoint', [probe, '--spec', a.spec, '--output', a.output],
           [a.spec, *map(Path, s['input_sha256'])],
           [Path(__file__), probe, SOURCE / 'finite_solve.py', SOURCE / 'finite_adapter.py',
            SOURCE / 'finite_units.py', SOURCE / 'numerical_acceptance.py',
            SOURCE / 'numerical_unit_guard.py', SOURCE / 'replay_steps.py'],
           [result, Path(str(result) + '.numerical.json'),
            a.output / 'profile.pstats', a.output / 'profile-summary.json'])
assert len(ledger.log) == 1
