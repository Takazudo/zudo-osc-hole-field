from unittest.mock import patch
p=Path(__file__).parent
source=Path('/home/takazudo/repos/circuits/zudo-osc-hole-field/worktrees/e16-campaign/design/partition/core-split/candidates/planning/joined-feed-construction-v2/finite_solve.py')
ns={'np':np,'_sparsetools':_sparsetools,'CAP':65536}
for path,names in [(source,['_extended_residual']),(p/'benchmark.py',['compiled_csc'])]:
 tree=ast.parse(path.read_text());nodes=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in names];exec(compile(ast.Module(body=nodes,type_ignores=[]),str(path),'exec'),ns)
old,new=ns['_extended_residual'],ns['compiled_csc'];results=[]
def compare(name,A,X,B):
 with np.errstate(invalid='ignore',over='ignore'):
  before=old(A,X,B);after=new(A,X,B)
 np.testing.assert_array_equal(before,after)
 np.testing.assert_array_equal(np.isnan(before),np.isnan(after))
 finite=np.isfinite(before)
 assert before[finite].tobytes()==after[finite].tobytes(),name+' signed zero/bit representation mismatch'
 results.append({'case':name,'status':'PASS','finite_residual_bytes_equal':True,'nonfinite_positions_equal':True})
# Width1 plus matrix columns; positive/negative zeros and negative coefficients.
for zero in [0.,-0.]:
 A=sp.csc_matrix((np.array([zero,-zero,3.,-2.]),np.array([1,0,1,0]),np.array([0,2,4])),shape=(2,2));compare('signed-zero-'+str(zero),A,np.array([zero,-zero],dtype=np.longdouble),np.array([-zero,zero]))
for av,xv in [(np.inf,0.),(np.nan,0.),(1.,np.nan),(1.,np.inf),(0.,np.inf)]:
 A=sp.csc_matrix(([av],([0],[0])),shape=(1,1));compare('nonfinite-'+str(av)+'-'+str(xv),A,np.array([xv],dtype=np.longdouble),np.array([1.]))
A=sp.csc_matrix([[100000001.,-100000000.],[-100000000.,100000000.]])
X=np.array([[1,1],[1+np.longdouble(1)/np.longdouble(1e8),1+np.longdouble(3)/np.longdouble(1e8)]],dtype=np.longdouble);compare('extended-cancellation',A,X,np.array([[0.,-2.],[1.,3.]]))
A=sp.csc_matrix((np.array([1e8,-1e8,1.,2.,-3.]),np.array([2,2,0,1,0]),np.array([0,4,5])),shape=(3,2));compare('unsorted-duplicates',A,np.ones((2,8),dtype=np.longdouble),np.zeros((3,8)))
A=sp.csc_matrix((np.ones(70001),(np.arange(70001),np.zeros(70001,dtype=int))),shape=(70001,1));compare('split-column',A,np.ones((1,2),dtype=np.longdouble)*np.longdouble('.123456789123456789'),np.ones((70001,2)))
A=sp.csc_matrix(([2.,-3.],([0,2],[0,70000])),shape=(3,70002));compare('empty-column-span',A,np.ones((70002,2),dtype=np.longdouble),np.zeros((3,2)))
compare('empty-matrix',sp.csc_matrix((4,5)),np.zeros((5,2)),np.zeros((4,2)))
# Spy existing compiled entry only; bounded scratch and stable single rr array.
A=sp.csc_matrix((np.ones(70001),(np.arange(70001),np.zeros(70001,dtype=int))),shape=(70001,1));calls=[];kernel=_sparsetools.csc_matvec

def bounded(*args):
 rows,cols,ptr,indices,data,vec,rr=args
 assert len(data)<=65536 and cols<=65536 and len(ptr)<=65537

 assert np.shares_memory(indices,A.indices)
 calls.append({'entries':len(data),'columns':cols,'rr_pointer':int(rr.ctypes.data),'data_bytes':data.nbytes,'vector_bytes':vec.nbytes})

with patch.object(_sparsetools,'csc_matvec',bounded):new(A,np.ones((1,1)),np.zeros((70001,1)))
assert len(calls)==2 and calls[0]['rr_pointer']==calls[1]['rr_pointer']
results.append({'case':'bounded-scratch-and-one-output-vector','status':'PASS','calls':calls})
