class PhaseCountTests(unittest.TestCase):
 def assess(self, counts, maximum=148):
  return module.phase_count_assessment(counts,24,111,12,maximum)
 def test_exact_full_baseline_counts_match(self):
  result=self.assess({'service':1,'reduce':24,'dpoint':111,'couple':12})
  self.assertTrue(result['complete'])
  self.assertEqual(result['diagnostics'][-1]['observed'],148)
 def test_extra_coupled_receipt_refused_even_with_room_under_ceiling(self):
  result=self.assess({'service':1,'reduce':24,'dpoint':111,'couple':13},maximum=200)
  self.assertFalse(result['complete'])
  coupled=next(d for d in result['diagnostics'] if d['phase']=='couple')
  self.assertEqual(coupled['status'],'EXCEEDED')
  self.assertEqual(coupled['expected'],12)
 def test_over_budget_run_refused_even_when_phase_counts_match(self):
  result=self.assess({'service':1,'reduce':24,'dpoint':111,'couple':12},maximum=147)
  self.assertFalse(result['complete'])
  self.assertEqual(result['diagnostics'][-1]['status'],'EXCEEDED')
 def test_unexpected_phase_refused(self):
  result=self.assess({'service':1,'reduce':24,'dpoint':111,'couple':12,'extra':1},maximum=200)
  self.assertFalse(result['complete'])
  self.assertIn({'phase':'extra','observed':1,'status':'UNEXPECTED_PHASE'},result['diagnostics'])
