"""Analytic 8-column extended refinement prototype; no production inputs."""
import ast
import copy
import hashlib
import json
from pathlib import Path
import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla
ROOT=Path(__file__).resolve().parents[2]
SOURCE=ROOT/'design/partition/core-split/candidates/planning/joined-feed-construction-v2/finite_solve.py'
tree=ast.parse(SOURCE.read_text())
helpers=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in ('_extended_residual','refined_solve')]
def need(ok,message):
    if not ok: raise ValueError(message)
ns={'np':np,'REFINE_STEPS':3,'need':need}
exec(compile(ast.Module(body=helpers,type_ignores=[]),str(SOURCE),'exec'),ns)
# Clone the actual numerical helper changing only its explicit width restriction;
# preserve the public helper above and every numerical operation/three corrections.
internal=copy.deepcopy(next(n for n in helpers if n.name=='refined_solve'))
internal.name='refined_eight'
for n in ast.walk(internal):
    if isinstance(n,ast.Call) and isinstance(n.func,ast.Name) and n.func.id=='need':
        n.args[0]=ast.parse('np.asarray(b).ndim == 1 or (np.asarray(b).ndim == 2 and b.shape[1] <= 8)',mode='eval').body
        n.args[1]=ast.Constant('prototype extended block requires at most eight RHS')
exec(compile(ast.fix_missing_locations(ast.Module(body=[internal],type_ignores=[])),str(SOURCE)+'-prototype-width-only','exec'),ns)
g=1e8;A=sp.csc_matrix([[g+1,-g],[-g,g]]);lu=spla.splu(A,permc_spec='MMD_AT_PLUS_A',diag_pivot_thresh=0.,options={'SymmetricMode':True})
W=np.array([[j/16 for j in range(16)],[1-j/16 for j in range(16)]])
W[:,7]=0 # Exercise zero column among nonzero supports.
W[:,8:]*=np.array([.5,2.,.001,3.,1.,1.5,.25,4.])
B=np.array([[-1.,0.,2.],[0.,-1.,-2.]])
original,oldres=ns['refined_solve'](lu,A,W)
e=np.empty(16);q=np.empty((3,16));absmax=0;independent=0;subresults=[]
for start in range(0,16,8):
    rhs=W[:,start:start+8]
    u,relative=ns['refined_eight'](lu,A,rhs,extended_solution=True)
    absmax=max(absmax,relative*max(float(np.max(abs(rhs))),1e-300))
    flow=np.longdouble(g)*(u[0]-u[1]);res=np.vstack([rhs[0]-u[0]-flow,rhs[1]+flow])
    independent=max(independent,float(np.max(abs(res))))
    e[start:start+8]=np.sum(rhs*u,axis=0);q[:,start:start+8]=B.T@u
    subresults.append({'start':start,'width':rhs.shape[1],'dtype':str(u.dtype),'relative':relative})
    del u
scale=max(float(np.max(abs(W))),1e-300);reported=absmax/scale
# Analytic x0=b0+b1; x1=x0+b1/g, evaluated independently in extended arithmetic.
Wext=W.astype(np.longdouble); x0=Wext.sum(axis=0); expected=np.vstack([x0,x0+Wext[1]/np.longdouble(g)])
expected_e=np.asarray(np.sum(Wext*expected,axis=0),dtype=np.float64);expected_q=np.asarray(B.astype(np.longdouble).T@expected,dtype=np.float64)
assert oldres>1e-9 and reported<1e-9 and independent/scale<1e-9
np.testing.assert_allclose(e,expected_e,rtol=1e-12,atol=1e-15)
np.testing.assert_allclose(q,expected_q,rtol=1e-12,atol=1e-15)
# Original single-extended-RHS public boundary remains intact.
try:ns['refined_solve'](lu,A,W,extended_solution=True)
except ValueError:public_refusal=True
else:public_refusal=False
assert public_refusal
z,zres=ns['refined_eight'](lu,A,np.zeros((2,8)),extended_solution=True)
assert not z.any() and zres==0
# Empty blocks should be skipped by the projection orchestration (max of empty
# is not defined in the existing helper); do not call helper on zero-column arrays.
report={'kind':'analytic_only_extended_subbatch','source_sha256':hashlib.sha256(SOURCE.read_bytes()).hexdigest(),'scheduling_batch':16,'extended_width':8,'refinement_steps':3,'old_residual':oldres,'candidate_whole_original_batch_normalized_residual':reported,'independent_branch_residual':independent/scale,'e_dtype':str(e.dtype),'q_dtype':str(q.dtype),'max_e_error':float(np.max(abs(e-expected_e))),'max_q_error':float(np.max(abs(q-expected_q))),'public_multicolumn_extended_refused':public_refusal,'zero_rhs_block_residual':zres,'subblocks':subresults,'array_bytes_per_node':{'common_rhs_batch_16_float64':16*8,'old_solution_16_float64':16*8,'candidate_solution_8_longdouble':8*np.dtype(np.longdouble).itemsize,'old_best_16_float64':16*8,'candidate_best_8_longdouble':8*np.dtype(np.longdouble).itemsize,'old_residual_16_float64':16*8,'candidate_residual_8_float64':8*8,'old_correction_16_float64':16*8,'candidate_correction_8_float64':8*8}}
print(json.dumps(report,indent=2))
