tree=ast.parse(source.read_text());node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='_extended_residual');ns={'np':np};exec(compile(ast.Module(body=[node],type_ignores=[]),str(source),'exec'),ns);original=ns['_extended_residual']
CAP=65536

def compiled_csc(A,x,b):
    assert A.format=='csc'
    xx=np.asarray(x);bb=np.asarray(b);vector=bb.ndim==1
    if vector:xx=xx[:,None];bb=bb[:,None]
    result=np.empty(bb.shape,dtype=np.float64)
    for j in range(bb.shape[1]):
        rr=bb[:,j].astype(np.longdouble)
        # Bound both column-pointer and data scratch, including empty columns
        # and individual columns larger than the data chunk.
        for c0 in range(0,A.shape[1],CAP):
            c1=min(c0+CAP,A.shape[1]);low=int(A.indptr[c0]);stop=int(A.indptr[c1])
            for lo in range(low,stop,CAP):
                hi=min(lo+CAP,stop)
                first=max(c0,int(np.searchsorted(A.indptr,lo,side='right'))-1)
                last=min(c1-1,int(np.searchsorted(A.indptr,hi-1,side='right'))-1)
                ptr=np.clip(A.indptr[first:last+2]-lo,0,hi-lo)
                data=-A.data[lo:hi].astype(np.longdouble)
                vec=xx[first:last+1,j].astype(np.longdouble)
                _sparsetools.csc_matvec(A.shape[0],last-first+1,ptr,A.indices[lo:hi],data,vec,rr)
        result[:,j]=rr
    return result[:,0] if vector else result

def timing(f,*args):
    t=time.perf_counter();v=f(*args);return v,time.perf_counter()-t
rows=[]
for n in [2000,20000,100000]:
    # Sparse SPD graph, finite shunts; no board or project input geometry.
    A=sp.diags([-np.ones(n-1)*1e5,np.ones(n)*(2e5+.1),-np.ones(n-1)*1e5],[-1,0,1],format='csc')
    A+=sp.diags([-np.ones(n-37)*13,np.ones(n)*26,-np.ones(n-37)*13],[-37,0,37],format='csc');A.sort_indices()
    rng=np.random.default_rng(42);X=rng.random((n,8)).astype(np.longdouble);B=rng.random((n,8))
    old,to=timing(original,A,X,B);new,tn=timing(compiled_csc,A,X,B)
    np.testing.assert_array_equal(new,old)
    lu,tf=timing(spla.splu,A);_,tl=timing(lu.solve,B);_,tp=timing(lambda:np.sum(B*X,axis=0))
    rows.append({'nodes':n,'nnz':A.nnz,'rhs':8,'old_residual_seconds':to,'compiled_residual_seconds':tn,'speedup':to/tn,'exact_float64_output_equal':True,'factor_seconds':tf,'one_LU_solve_seconds':tl,'one_e_projection_seconds':tp})

]:

    o=original(A,x,b);v=compiled_csc(A,x,b);np.testing.assert_array_equal(o,v);cases.append({'case':name,'exact_output_equal':True})
