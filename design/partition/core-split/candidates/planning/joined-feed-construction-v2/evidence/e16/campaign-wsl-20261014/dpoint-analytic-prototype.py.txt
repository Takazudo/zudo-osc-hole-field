"""Tiny analytic driving-point prototype; no production geometry or pickle input."""
import ast
import hashlib
import json
from pathlib import Path
import sys
import time
import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla

ROOT=Path(__file__).resolve().parents[2]
SOURCE=ROOT/'design/partition/core-split/candidates/planning/joined-feed-construction-v2/finite_solve.py'
tree=ast.parse(SOURCE.read_text())
helpers=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in ('_extended_residual','refined_solve')]
def need(ok,message):
    if not ok: raise ValueError(message)
ns={'np':np,'REFINE_STEPS':3,'need':need}
exec(compile(ast.Module(body=helpers,type_ignores=[]),str(SOURCE),'exec'),ns)
# One 1S source shunt at node0, one highly conductive 1e8S link.
# Sixteen normalized nonnegative unit-current supports test driving points.
g=100000000.
A=sp.csc_matrix([[g+1,-g],[-g,g]])
W=np.array([[j/16 for j in range(16)],[1-j/16 for j in range(16)]])
B=np.array([[-1.],[0.]])
lu=spla.splu(A,permc_spec='MMD_AT_PLUS_A',diag_pivot_thresh=0.,options={'SymmetricMode':True})
U,reported=ns['refined_solve'](lu,A,W)
old_e=np.sum(W*U,axis=0);old_q=B.T@U
# Candidate projects each extended solution before casting only e/q, not U.
# At most one extended node column and its best-iterate scratch; W stays16-wide.
e=np.empty(16,dtype=np.float64);q=np.empty((1,16),dtype=np.float64)
relative=[];independent=[];rounded=[];column_dtypes=[]
for j in range(16):
    x,res=ns['refined_solve'](lu,A,W[:,j],extended_solution=True)
    column_dtypes.append(str(x.dtype));relative.append(res)
    flow=np.longdouble(g)*(x[0]-x[1])
    r=np.array([np.longdouble(W[0,j])-x[0]-flow,np.longdouble(W[1,j])+flow])
    independent.append(float(np.max(abs(r))/np.max(abs(W[:,j]))))
    rounded.append(float(np.max(abs(ns['_extended_residual'](A,x.astype(np.float64),W[:,j])))/np.max(abs(W[:,j]))))
    e[j]=W[:,j]@x;q[:,j]=B.T@x
# Exact solution x0=1, x1=1+b1/g. e=1+b1^2/g, q=-1.
expected_e=1+W[1]**2/g
expected_q=-np.ones((1,16))
report={'kind':'analytic_source_fixture_not_production_diagnosis','source_sha256':hashlib.sha256(SOURCE.read_bytes()).hexdigest(),'matrix':A.toarray().tolist(),'batch':16,'refinement_steps':3,'original_dtype':str(U.dtype),'original_reported_residual':reported,'candidate_worst_column_normalized_residual':max(relative),'candidate_independent_branch_residual':max(independent),'rounding_back_to_float64_worst_residual':max(rounded),'candidate_e_dtype':str(e.dtype),'candidate_q_dtype':str(q.dtype),'candidate_max_e_error':float(np.max(abs(e-expected_e))),'candidate_max_q_error':float(np.max(abs(q-expected_q))),'candidate_max_difference_from_old_e':float(np.max(abs(e-old_e))),'candidate_max_difference_from_old_q':float(np.max(abs(q-old_q))),'extended_columns_live_at_once':1,'per_column_residuals':relative}
assert reported>1e-9,report
assert max(relative)<1e-9 and max(independent)<1e-9,report
assert max(rounded)>1e-9,report
np.testing.assert_allclose(e,expected_e,rtol=1e-12,atol=0)
np.testing.assert_allclose(q,expected_q,rtol=1e-12,atol=0)
print(json.dumps(report,indent=2))
